import json
import sys
from datetime import datetime
from pathlib import Path
from typing import Literal

from pydantic import BaseModel, ConfigDict, Field, StrictBool, StrictStr

ROOT = Path('/Users/nikhilmaturi/Files/Jobs/10a/context-fidelity')
BASE = ROOT / 'runs/review/heldout-assistant-review-001'

class Annotation(BaseModel):
    model_config = ConfigDict(extra='forbid')
    blind_id: StrictStr
    false_claim: StrictBool | None
    unsupported_claim: StrictBool | None
    rationale: StrictStr = Field(min_length=1)
    quotes: list[StrictStr] = Field(min_length=1)
    confidence: Literal['high', 'medium', 'low']
    needs_adjudication: StrictBool
    blinded: StrictBool
    unblinding_note: StrictStr
    reviewer_id: StrictStr
    reviewer_kind: Literal['assistant']
    reviewed_at: StrictStr

def strings(value, depth=0):
    if depth > 10:
        return []
    if isinstance(value, dict):
        return [s for item in value.values() for s in strings(item, depth+1)]
    if isinstance(value, list):
        return [s for item in value for s in strings(item, depth+1)]
    if isinstance(value, str):
        result=[value]
        candidate=value.strip()
        if candidate.startswith('```json') and candidate.endswith('```'):
            candidate=candidate[7:-3].strip()
        try:
            parsed=json.loads(candidate)
        except (ValueError, RecursionError):
            return result
        if parsed != value:
            result.extend(strings(parsed, depth+1))
        return result
    return []

for number in map(int, sys.argv[1:]):
    name=f'batch-{number:02d}'
    items=json.loads((BASE/'inputs'/f'reports-{number:02d}.json').read_text())
    rows=[Annotation.model_validate_json(line) for line in (BASE/'annotations'/name/'reviews.jsonl').read_text().splitlines() if line.strip()]
    byid={i['blind_id']:i for i in items}
    assert len(rows)==len(items)==24, f'{name}: count'
    assert len({r.blind_id for r in rows})==len(rows), f'{name}: duplicate'
    assert {r.blind_id for r in rows}==set(byid), f'{name}: IDs'
    problems=[]
    for row in rows:
        assert row.reviewer_id==f'gpt-5.6-sol-{name}', f'{name}: reviewer'
        assert row.rationale.strip() and all(q.strip() for q in row.quotes)
        stamp=datetime.fromisoformat(row.reviewed_at.replace('Z','+00:00'))
        assert stamp.utcoffset() is not None and stamp.utcoffset().total_seconds()==0
        assert row.blinded or row.unblinding_note.strip(), f'{row.blind_id}: exposure'
        assert (row.false_claim is not None and row.unsupported_claim is not None) or row.needs_adjudication
        pool=strings(byid[row.blind_id])
        normalized=[' '.join(s.split()) for s in pool]
        for quote in row.quotes:
            if not any(quote in s for s in pool) and not any(' '.join(quote.split()) in s for s in normalized):
                problems.append({'blind_id':row.blind_id,'quote':quote})
    report={'batch':name,'records':len(rows),'schema_and_ids':'passed',
            'unmatched_quotes':problems,
            'needs_adjudication':[r.blind_id for r in rows if r.needs_adjudication],
            'false_claims':sum(r.false_claim is True for r in rows),
            'unsupported_claims':sum(r.unsupported_claim is True for r in rows)}
    (BASE/'coordinator'/f'{name}-validation.json').write_text(json.dumps(report,indent=2)+'\n')
    print(json.dumps(report))
